function decodePNG(bytes: Uint8Array) { const png=Buffer.from(bytes),idat:Buffer[]=[];let offset=8;const width=png.readUInt32BE(16),height=png.readUInt32BE(20);while(offset<png.length){const n=png.readUInt32BE(offset);if(png.toString('ascii',offset+4,offset+8)==='IDAT')idat.push(png.subarray(offset+8,offset+8+n));offset+=n+12;}const raw=inflateSync(Buffer.concat(idat)),rgba=Buffer.alloc(width*height*4);for(let y=0;y<height;y++){expect(raw[y*(width*4+1)]).toBe(0);raw.copy(rgba,y*width*4,y*(width*4+1)+1,(y+1)*(width*4+1));}return{width,height,rgba}; }
